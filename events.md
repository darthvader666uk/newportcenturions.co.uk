<script>
/* Club calendar: show one month at a time, and open an event on click.

   Progressive enhancement throughout. With JavaScript off, every month stays
   on the page and the agenda carries the full text of each entry, which is
   also exactly what a crawler sees. Nothing here is required to read the
   calendar; it only makes a long page shorter and the detail easier to reach.

   The event schema in the head lists every upcoming entry regardless, so
   paging the view never hides anything from search engines. */
(function () {
  var payload = document.getElementById('cal-data');
  var list = document.querySelector('.fx-list');
  if (!payload || !list) return;

  var events = {};
  try {
    JSON.parse(payload.textContent).forEach(function (e) { events[e.id] = e; });
  } catch (err) {
    return;  // Malformed data: leave the page as the server rendered it.
  }

  // Defined below, once the dialog is set up; the pager only calls it on click.
  var closeDetail = function () {};

  /* ---- One month at a time -------------------------------------------- */

  var pager = document.querySelector('.cal-pager');
  var label = document.querySelector('.cal-pager-label');
  var prev = document.querySelector('[data-cal-prev]');
  var next = document.querySelector('[data-cal-next]');
  var months = [].slice.call(list.querySelectorAll('.fx-month'));

  if (pager && months.length) {
    var keys = months.map(function (s) { return s.getAttribute('data-month'); });

    // Start on the real current month, not the one that was current at build
    // time. A cached page could otherwise open on a month that has passed.
    var now = new Date();
    var todayKey = now.getFullYear() + '-' + ('0' + (now.getMonth() + 1)).slice(-2);
    var at = keys.indexOf(todayKey);
    if (at < 0) {
      // No month for today. Fall forward to the next one that has something in
      // it. Starting from the last month rather than the first matters when
      // every month on the page is in the past, which happens to a page cached
      // past the end of the data: showing the most recent month beats opening
      // on one from a year ago.
      at = keys.length - 1;
      for (var i = 0; i < keys.length; i++) {
        if (keys[i] >= todayKey) { at = i; break; }
      }
    }

    var show = function (i) {
      at = Math.max(0, Math.min(months.length - 1, i));
      months.forEach(function (s, n) { s.hidden = n !== at; });
      label.textContent = months[at].querySelector('.fx-month-label').textContent;
      prev.disabled = at === 0;
      next.disabled = at === months.length - 1;
    };

    // Whatever the popover was pinned to is about to be hidden.
    prev.addEventListener('click', function () { closeDetail(); show(at - 1); });
    next.addEventListener('click', function () { closeDetail(); show(at + 1); });

    list.classList.add('is-paged');
    pager.hidden = false;
    show(at);
  }

  /* ---- Event details --------------------------------------------------- */

  var dialog = document.getElementById('cal-detail');
  if (!dialog || typeof dialog.showModal !== 'function') return;

  /* The page wrapper carries a backdrop-filter, which makes it the containing
     block for anything position:fixed inside it, so a non-modal popover would
     be laid out against the wrapper rather than the window. A modal dialog
     escapes this through the top layer; this one has to be moved out. */
  document.body.appendChild(dialog);

  var field = function (name) { return dialog.querySelector('[data-f="' + name + '"]'); };

  /* Draw a team's badge, or its initials when we have no crest on file. Mirrors
     what _includes/team-crest.html does on the cards, so the pop-up and the
     card behind it never disagree. */
  var badge = function (slot, url, slug) {
    slot.textContent = '';
    slot.className = 'cal-dialog-crest';
    if (url) {
      var img = document.createElement('img');
      img.src = url;
      img.alt = '';
      img.width = 34;
      img.height = 34;
      slot.appendChild(img);
      return;
    }
    var parts = (slug || '').split('-');
    slot.className = 'cal-dialog-crest cal-dialog-crest--mono';
    slot.textContent = (parts[0] || '').slice(0, 1).toUpperCase()
      + (parts[1] ? parts[1].slice(0, 1).toUpperCase() : '');
  };

  var GAP = 10;   // breathing room between the entry and the popover
  var EDGE = 8;   // keep this far off the edge of the window

  /* Put the popover beside the entry, like a calendar app: to the right if it
     fits, otherwise to the left, otherwise centred. Vertically it lines up with
     the entry and then gets pushed back inside the window if it would overflow.
     Measured after opening, because a closed dialog has no dimensions. */
  var place = function (anchor) {
    var a = anchor.getBoundingClientRect();
    var d = dialog.getBoundingClientRect();
    var vw = document.documentElement.clientWidth;
    var vh = document.documentElement.clientHeight;

    var left;
    if (a.right + GAP + d.width <= vw - EDGE) {
      left = a.right + GAP;
    } else if (a.left - GAP - d.width >= EDGE) {
      left = a.left - GAP - d.width;
    } else {
      left = Math.max(EDGE, (vw - d.width) / 2);
    }

    var top = Math.min(
      Math.max(EDGE, a.top - 8),
      Math.max(EDGE, vh - d.height - EDGE)
    );

    dialog.style.left = Math.round(left) + 'px';
    dialog.style.top = Math.round(top) + 'px';
  };

  var anchored = false;
  var current = null;   // the element the popover is currently pinned to

  var open = function (id, anchor) {
    var e = events[id];
    if (!e) return;

    // Wide enough to sit the popover next to the entry and still see the month.
    anchored = window.matchMedia('(min-width: 760px)').matches;
    dialog.className = 'cal-dialog cat-' + e.cat + (anchored ? ' cal-dialog--anchored' : '');
    dialog.style.left = '';
    dialog.style.top = '';

    field('catLabel').textContent = e.catLabel || '';
    field('trophy').hidden = !e.trophy;

    // A fixture shows both badges and the two sides; anything else just its
    // title. The <h2> always ends up with text either way, which matters
    // because it is what aria-labelledby points the dialog at.
    var fixture = !!(e.homeTeam && e.awayTeam);
    field('teams').hidden = !fixture;
    field('title').hidden = fixture;
    if (fixture) {
      field('homeTeam').textContent = e.homeTeam;
      field('awayTeam').textContent = e.awayTeam;
      badge(field('homeCrest'), e.homeCrest, e.homeSlug);
      badge(field('awayCrest'), e.awayCrest, e.awaySlug);
    }
    field('title').textContent = e.title;

    var when = e.when;
    if (!e.allDay && e.time) {
      when += ', ' + e.time + (e.endTime ? ' to ' + e.endTime : '');
    } else if (e.allDay) {
      when += ', all day';
    }
    if (e.side) when += ' (' + e.side + ')';
    field('when').textContent = when;

    var whereLine = field('whereLine');
    var map = field('map');
    whereLine.hidden = !e.where;
    if (e.where) {
      map.textContent = e.where;
      map.href = 'https://www.google.com/maps/search/?api=1&query=' + encodeURIComponent(e.where);
    }

    var desc = field('desc');
    desc.textContent = e.desc || '';
    desc.hidden = !e.desc;

    if (anchored) {
      current = anchor;
      // Opening moves focus to the dialog, and the browser scrolls to wherever
      // that is. Park it next to the entry first so there is nothing to scroll
      // to, then measure and place it properly.
      var a = anchor.getBoundingClientRect();
      dialog.style.left = Math.round(a.right) + 'px';
      dialog.style.top = Math.round(a.top) + 'px';
      dialog.show();       // non-modal, so the month behind stays readable
      place(anchor);
    } else {
      current = null;
      dialog.showModal();  // phone: centred, with a backdrop
    }
  };

  var close = function () {
    current = null;
    if (dialog.open) dialog.close();
  };
  closeDetail = close;

  document.addEventListener('click', function (ev) {
    var target = ev.target;
    if (!target || !target.closest) return;

    var trigger = target.closest('[data-ev]');
    if (trigger) { open(trigger.getAttribute('data-ev'), trigger); return; }

    if (target.closest('[data-cal-close]')) { close(); return; }
    if (target === dialog) { close(); return; }   // modal backdrop
    // A non-modal dialog gets no backdrop, so dismiss it on any outside click.
    if (anchored && dialog.open && !target.closest('.cal-dialog')) close();
  });

  // Modal dialogs handle Escape themselves; non-modal ones do not.
  document.addEventListener('keydown', function (ev) {
    if (ev.key === 'Escape' && anchored) close();
  });

  // A popover pinned to an entry has to follow it when the page moves, rather
  // than close: opening one scrolls the page itself, so closing on scroll shuts
  // the popover the instant it appears.
  var pending = false;
  var reflow = function () {
    if (!current || !dialog.open || pending) return;
    pending = true;
    requestAnimationFrame(function () {
      pending = false;
      if (current && dialog.open) place(current);
    });
  };
  window.addEventListener('scroll', reflow, true);
  window.addEventListener('resize', reflow);
}());
</script>
